How to Embed in 5 Lines of Code 

<script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
<script src="assets/js/cast-graph-engine.js"></script>
<div id="myGraphContainer" style="height:400px;"></div>

<script>
  const graph = new CASTGraphEngine('#myGraphContainer', {
    title: "Heating Curve of Water",
    xAxis: { label: "Heating Time", unit: "min" },
    yAxis: { label: "Temperature", unit: "°C" },
    chartType: "scatter",
    showBestFit: true,
    xValues: [0, 2, 4, 6, 8, 10],
    series: [{ label: "Sample A", values: [0, 20, 40, 60, 80, 100], color: "#0284c7" }]
  });
</script>
